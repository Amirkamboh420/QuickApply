   259→function Stat({ value, suffix, label, decimals = 0 }: { value: number; suffix?: string; label: string; decimals?: number }) {
   260→  const [display, setDisplay] = useState(0);
   261→  const ref = useRef<HTMLDivElement>(null);
   262→  const started = useRef(false);
   263→
   264→  useEffect(() => {
   265→    const el = ref.current;
   266→    if (!el) return;
   267→    const obs = new IntersectionObserver(
   268→      (entries) => {
   269→        if (entries[0].isIntersecting && !started.current) {
   270→          started.current = true;
   271→          const duration = 1400;
   272→          const start = performance.now();
   273→          const tick = (now: number) => {
   274→            const p = Math.min((now - start) / duration, 1);
   275→            // ease-out cubic
   276→            const eased = 1 - Math.pow(1 - p, 3);
   277→            setDisplay(value * eased);
   278→            if (p < 1) requestAnimationFrame(tick);
   279→          };
   280→          requestAnimationFrame(tick);
   281→        }
   282→      },
   283→      { threshold: 0.4 },
   284→    );
   285→    obs.observe(el);
   286→    return () => obs.disconnect();
   287→  }, [value]);
   288→
   289→  const formatted = display.toLocaleString("en-US", {
   290→    minimumFractionDigits: decimals,
   291→    maximumFractionDigits: decimals,
   292→  });
   293→
   294→  return (
   295→    <div ref={ref}>
   296→      <div className="text-2xl font-bold text-foreground tabular-nums">
   297→        {formatted}
   298→        <span className="text-ocean">{suffix}</span>
   299→      </div>
   300→      <div className="text-xs text-muted-foreground mt-0.5">{label}</div>
   301→    </div>
   302→  );
   303→}

   475→function LiveActivity() {
   476→  const [activities, setActivities] = useState<ActivityItem[]>(FALLBACK_ACTIVITIES);
   477→  const [visible, setVisible] = useState(0);
   478→  const [connected, setConnected] = useState(false);
   479→
   480→  useEffect(() => {
   481→    // Try to connect to the live WebSocket mini-service
   482→    let socket: { close: () => void } | null = null;
   483→    let interval: ReturnType<typeof setInterval> | null = null;
   484→    let cancelled = false;
   485→
   486→    try {
   487→      const io = (window as unknown as { io?: unknown }).io;
   488→      if (typeof io === "function") {
   489→        // socket.io client would be loaded via script; here we use the global
   490→      }
   491→    } catch {
   492→      // ignore
   493→    }
   494→
   495→    // Use a lightweight approach: dynamically import socket.io-client if available,
   496→    // otherwise fall back to interval rotation of fallback data.
   497→    import("socket.io-client")
   498→      .then(({ io }) => {
   499→        if (cancelled) return;
   500→        socket = io("/?XTransformPort=3004", { path: "/", transports: ["websocket", "polling"] });
   501→        (socket as unknown as { on: (e: string, cb: (...a: unknown[]) => void) => void }).on("connect", () => setConnected(true));
   502→        (socket as unknown as { on: (e: string, cb: (...a: unknown[]) => void) => void }).on("disconnect", () => setConnected(false));
   503→        (socket as unknown as { on: (e: string, cb: (...a: unknown[]) => void) => void }).on("recent-activities", (acts: unknown) => {
   504→          if (Array.isArray(acts) && acts.length) {
   505→            setActivities(acts as ActivityItem[]);
   506→            setVisible(0);
   507→          }
   508→        });
   509→        (socket as unknown as { on: (e: string, cb: (...a: unknown[]) => void) => void }).on("new-activity", (act: unknown) => {
   510→          setActivities((prev) => [act as ActivityItem, ...prev].slice(0, 10));
